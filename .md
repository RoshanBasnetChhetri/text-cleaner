// build step 1: basic cleaning function

export function cleanText(text:string):string{
    return text.trim()
}

// This is the simple version of our engine.
// It only removes extra spaces at the start and end



//Upgrade to real engine (add rules system)
//step 2: add options (Engineering Thinking)

export type CleanOptions = {
    removeExtraSpaces?: boolean;
    removeLineBreak?: boolean;
    removeSpecialChars?:boolean;
}
//here we are not hardcoding behavior. we are makign the system configurable




"We create let result = text because text passes through multiple cleaning stages. Each stage takes the output from the previous stage and transforms it again. Think of it like a car assembly line: the same car moves through different stations and gets improved at each step. The result variable represents the current state of the text as it moves through our cleaning pipeline."
